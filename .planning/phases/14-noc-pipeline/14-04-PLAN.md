From components.jsx — answerValid (current):
```jsx
export function answerValid(c, value) {
  if (c.type === 'text' || c.type === 'textarea') return !!(value || '').trim();
  if (c.type === 'choices') return !!value;
  if (c.type === 'scale') return value !== null && value !== undefined;
  if (c.type === 'duties') return Array.isArray(value) && value.length > 0;
  if (c.type === 'drf') return !!value;
  if (c.type === 'qual') return !!(value && value.education && value.experience);
  return false;
}
```
