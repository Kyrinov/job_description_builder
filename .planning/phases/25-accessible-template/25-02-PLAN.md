OUTPUT_PATH = "v2/backend/app/templates/wd_template.docx"   # -> change to wd_accessible_template.docx

def _set_cell_text(cell, text, *, bold=False, italic=False):
    for para in list(cell.paragraphs):
        p_el = para._p
        p_el.getparent().remove(p_el)
    para = cell.add_paragraph()
    run = para.add_run(text)
    if bold: run.bold = True
    if italic: run.italic = True

# {%p for %} paragraph loop (own paragraph per tag):
doc.add_paragraph("{%p for duty in duties %}")
doc.add_paragraph("{{ duty.text }}")
doc.add_paragraph("{%p endfor %}")

# {%tr for %} table-row loop — tag ALONE in first cell of its own row:
_set_cell_text(t.rows[1].cells[0], "{%tr for f in effort_factors %}")
_set_cell_text(t.rows[2].cells[0], "{{ f.factor_name }}")
_set_cell_text(t.rows[3].cells[0], "{%tr endfor %}")
# PITFALL (preserve verbatim): docxtpl's patch_xml regex is greedy and matches the
# LAST {%tr %} tag in a row — never co-locate for/endfor, never share a row with data.
