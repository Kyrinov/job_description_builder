    <p class="jes-total-points">Total Points: <strong>{{ jes_total_points }}</strong></p>

    {% for score in jes_scores %}

            {% endif %}
        </div>
        <p class="jes-rationale">{{ score.rationale }}</p>
        <p class="jes-source muted">
            Source: {{ score.provenance.source_id }} ({{ score.provenance.source_version }})
