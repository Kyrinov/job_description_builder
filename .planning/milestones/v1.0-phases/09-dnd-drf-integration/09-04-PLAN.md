                    <strong>{{ c.core_responsibility }}</strong>:
                    {{ c.departmental_result }}
                    <span class="drf-fiscal-year">({{ c.fiscal_year }})</span>
                    <span class="drf-score-badge">{{ c.score }} match{{ "es" if c.score != 1 else "" }}</span>
                </label>
            </li>
        {% endfor %}
        </ul>
        <!-- row_ids is a comma-joined string of selected checkbox values, built by Alpine or JS -->

            type="submit"
            class="button button--primary"
            onclick="
                const checked = document.querySelectorAll('input[name=\"candidate_ids[]\"]:checked');
