```python
@router.post("/wd/{wd_id}/confirm-subgroup")
async def confirm_subgroup(wd_id: str, body: SubGroupConfirmRequest) -> dict:
    """POST /api/wd/{id}/confirm-subgroup — store confirmed sub-group on WorkDescription.

    T-21-01: validates sub_group against ALLOWED_SUBGROUPS[og_code] before storage.
    Returns 422 if sub_group is not in the allowed set for the WD's confirmed_og.
    """
    # Read current WD from DB
    # (use same DB access pattern as existing /api/wd/{id} endpoints in the codebase)
    wd = _get_wd(wd_id)  # or await _get_wd(wd_id) — match existing async pattern
    if wd is None:
        raise HTTPException(status_code=404, detail=f"WorkDescription {wd_id!r} not found")

    # Determine og_code from confirmed_og (may be str or dict)
    confirmed_og = wd.confirmed_og
    if isinstance(confirmed_og, dict):
        og_code = confirmed_og.get("og_code", "")
    else:
        og_code = confirmed_og or ""

    # T-21-01: validate sub_group against allowed set
    if og_code in ALLOWED_SUBGROUPS:
        allowed = ALLOWED_SUBGROUPS[og_code]
        if body.sub_group not in allowed:
            raise HTTPException(
                status_code=422,
                detail={
                    "error": "invalid_sub_group",
                    "message": f"sub_group {body.sub_group!r} is not valid for OG {og_code!r}",
                    "allowed_values": sorted(allowed),
                },
            )

    # Persist confirmed_sub_group on WD
    wd.confirmed_sub_group = body.sub_group
    _save_wd(wd_id, wd)  # match existing WD save pattern
    return {"status": "ok", "confirmed_sub_group": body.sub_group}
```
